"""来源：62 案例.py；配套：62 案例-优化版.md。

相关改写的原因和行为差异见就近中文说明。

from math import isfinite


def triangle_area(base, height):
    """根据底和高返回三角形面积。"""
    # 原Triangle_area首字母大写；改snake_case，base比字母l更说明底的含义。
    return base * height / 2


def alpha_count(text):
    """统计本题限定的英语元音a/e/i/o/u，包含大写。"""
    # 原str遮蔽内置类型；原for/if累加改sum消费生成器，True算1，False算0。
    # 这是单次遍历的简洁写法，不声称比原循环实测更快。
    return sum(character in "aeiouAEIOU" for character in text)


def score_analyse(score_list: list[float]):
    """返回最高分、最低分、舍入一位小数的平均分。类型注解不自动验证。"""
    # 原max/min空表会ValueError；给出明确前置条件，不伪造空表平均数。
    if not score_list:
        raise ValueError("成绩列表不能为空")
    return max(score_list), min(score_list), round(sum(score_list) / len(score_list), 1)


def main():
    # 原文件全部注释，当前不会执行；本优化版启用三个课堂案例，属于行为变化。
    print(triangle_area(1, 2))
    print(f"传入的字符串中元音字母的个数{alpha_count(input('请输入字符串: '))}")
    scores = []
    while True:
        value = input("请输入高考成绩(输入'end'结束): ")
        if value == "end":
            if not scores:
                print("尚未录入成绩，请先输入至少一项")

            highest, lowest, average = score_analyse(scores)
            print(f"最高分{highest} 最低分{lowest} 平均分{average:.1f}")
            break
        try:
            score = float(value)
        except ValueError:
            print("请输入有效的数字，或输入 end 结束")
        else:
            # 新边界：拒绝nan/inf；没有擅自把高考成绩上限设为100。
            if isfinite(score):
                scores.append(score)
            else:

"""来源：练习.py；配套：62 案例-优化版.md。

相关改写的原因和行为差异见就近中文说明。

from math import isfinite


def score_grade(score: float):
    """按题目90/75/60三个阈值分类；题目未规定上下限。"""
    # 原elif/else均在前一个return之后；早返回可去掉额外嵌套，阈值和顺序不变。
    if score >= 90:
        return "A"
    if score >= 75:
        return "B"
    if score >= 60:
        return "C"
    return "D"


def is_palindrome(text: str) -> bool:
    """直接比较反转串；空串按该定义为True，不忽略标点。"""
    # 原Palindrome(str)遮蔽str，且返回文字，与题目bool要求不符。
    # return text == text[::-1]消除可由比较直接得到的if/else，返回类型改为bool。
    return text == text[::-1]


def time_trans(seconds: int):
    """将非负整数秒转换为时／分／秒；保留原来的分段显示风格。"""
    if type(seconds) is not int or seconds < 0:
        raise ValueError("秒数须为非负整数")
    # 原hour*60混用了小时和分钟单位，尾部%3600也不能得到秒字段。
    # 新divmod(a,b)同时返回商与余数：先分出小时，再从余秒分出分钟和秒。
    hour, remaining_seconds = divmod(seconds, 3600)
    minute, second = divmod(remaining_seconds, 60)
    if hour:
        return f"{hour}时 {minute}分 {second}秒"
    if minute:
        return f"{minute}分{second}秒"
    return f"{second}秒"


def triangle_judge(a, b, c):
    """有限正边长且任意两边之和大于第三边；等边优先于等腰。"""
    # 原or已修正，不能再称当前用了|；仍只检查a+b<=c，漏其他排列与非正数。
    if not all(isfinite(side) and side > 0 for side in (a, b, c)):
        return "不能构成三角形"
    if a + b <= c or a + c <= b or b + c <= a:
        return "不能构成三角形"
    if a == b == c:
        return "等边三角形"
    if a == b or a == c or b == c:
        return "等腰三角形"
    return "普通三角形"


if __name__ == "__main__":
    for value in (90, 80, 65, 45):
        print(score_grade(value))

        print(time_trans(seconds))
