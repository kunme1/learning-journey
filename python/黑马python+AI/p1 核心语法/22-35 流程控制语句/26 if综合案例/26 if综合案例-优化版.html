dianliang = int(input("请输入电量："))

if dianliang <2880:
    print(f"电费是 {dianliang * 0.4883}")
elif 2880 <= dianliang <= 4800:
    print(f"电费是 {2880 * 0.4883 + (dianliang - 2880) * 0.5383}")
elif 4800 < dianliang:
    print(f"电费是 {2880 * 0.4883 + (4800 - 2880) * 0.5383 +(dianliang - 4800) * 0.7883}")
else:
    print("输入的电量不合理")
